\begin{frame}{Unified System Architecture}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \safeimage[width=\textwidth]{docs/images/demo_overview.png}{Architecture Overview}
      \vspace{0.1cm}
      \footnotesize\centering Multi-model FastAPI hub with dual client consumers
    \end{column}

    \begin{column}{0.52\textwidth}
      \begin{block}{Architectural Design Principles}
        \begin{itemize}
          \item \textbf{Decoupled Core:} \texttt{core/} contains purely algorithmic model classes, isolated from web frameworks.
          \item \textbf{Single Source of Truth:} \texttt{config.py} centralizes all settings with environment variable overrides.
          \item \textbf{Dynamic Lazy Loading:} \texttt{LOADERS} dictionary initializes only models specified in \texttt{ENABLED\_MODELS}.
          \item \textbf{Consistent API Contract:}
            \begin{itemize}
              \item Out-of-bounds parameter clamping (\texttt{top\_k}, \texttt{conf}, \texttt{k}).
              \item Standardized image format: \texttt{data:image/jpeg;base64,...}.
              \item Execution profiling via \texttt{X-Process-Time-ms} header.
            \end{itemize}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{AI-01: Flower Classification (ResNet-18)}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \safeimage[width=\textwidth]{docs/images/demo_classify.png}{Flower Classification}
    \end{column}

    \begin{column}{0.51\textwidth}
      \begin{block}{Model Architecture \& Dataset}
        \begin{itemize}
          \item \textbf{Model:} Fine-tuned ResNet-18 (transfer learning from ImageNet).
          \item \textbf{Dataset:} TF Flowers (3,670 images, 5 classes: daisy, dandelion, roses, sunflowers, tulips).
          \item \textbf{Stratified Split:} 80\% train, 10\% val, 10\% test.
        \end{itemize}
      \end{block}

      \begin{exampleblock}{Evaluation Metrics \& Inference}
        \begin{itemize}
          \item \textbf{Test Accuracy:} \textbf{94.8\%} (Target: $>88\%$).
          \item \textbf{Macro F1-Score:} \textbf{0.947}.
          \item \textbf{Safety Mechanism:} Returns \texttt{confident: false} if top prediction score is low (flags out-of-distribution inputs).
          \item \textbf{Inference Speed:} p50 = \textbf{157.7 ms} on standard CPU.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

    \begin{column}{0.46\textwidth}
      \safeimage[width=\textwidth]{docs/images/demo_search.png}{CLIP + FAISS Image Search}

    \begin{column}{0.51\textwidth}
      \begin{block}{Vector Search Architecture}
        \begin{itemize}
          \item \textbf{Encoder:} OpenAI CLIP (\texttt{ViT-B/32}) multi-modal embeddings ($512$-dim vector space).
          \item \textbf{Index:} FAISS \texttt{IndexFlatIP} with L2-normalized vectors (exact cosine similarity).
          \item \textbf{Gallery:} 500 cataloged flower images across 5 classes.

      \end{block}

      \begin{exampleblock}{Retrieval Precision \& Speed}
        \begin{itemize}
          \item \textbf{Text-to-Image P@10:} \textbf{100.0\%} precision at top-10.
          \item \textbf{Image-to-Image P@5:} \textbf{88.7\%} precision at top-5.
          \item \textbf{Search Latency:} p50 = \textbf{37.5 ms} (sub-50ms CPU search).
          \item \textbf{Flexible Queries:} Supports free-form text search or reverse image lookup.
        \end{itemize}

\begin{frame}{AI-04: Knowledge-Base Chatbot (RAG Architecture)}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{RAG Retrieval Pipeline}
        \begin{itemize}
          \item \textbf{Corpus:} 6 ShopLite e-commerce policy documents (\texttt{data/kb/*.md}).
          \item \textbf{Chunking:} 600-character overlapping chunks.
          \item \textbf{Embedding Model:} \\
            \texttt{paraphrase-multilingual-MiniLM-L12-v2}.
          \item \textbf{Generative Backbone:}
            \begin{itemize}
              \item Local: Qwen2.5-0.5B-Instruct (CPU/GPU).
              \item External: OpenAI-compatible API endpoint.
            \end{itemize}
          \item \textbf{Evaluation:} Hit@1 = \textbf{92.9\%}, Hit@3 = \textbf{100\%}.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{Safety Guardrails \& Streaming}
        \begin{itemize}
          \item \textbf{SSE Streaming Protocol:}
            \begin{itemize}
              \item \texttt{sources} event $\rightarrow$ token stream $\rightarrow$ \texttt{done}.
              \item Real-time conversational feel without blocking.
            \end{itemize}
          \item \textbf{Mandatory Citations:}
            \begin{itemize}
              \item System enforces strict \texttt{[file.md]} citations.
            \end{itemize}
          \item \textbf{Hallucination \& Out-of-Domain Guard:}
            \begin{itemize}
              \item Automated fallback response when query cannot be answered from context.
            \end{itemize}
          \item \textbf{Prompt-Injection Defense:}
            \begin{itemize}
              \item Context treated as passive data, not instructions.
            \end{itemize}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
