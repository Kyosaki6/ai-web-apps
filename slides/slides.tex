% SLIDE 6: AI-04 Chatbot RAG
% -------------------------------------------------------------
\slideheader{AI-04: Knowledge-Base Chatbot (RAG)}

\begin{minipage}[c]{0.48\textwidth}
  {\large RAG Pipeline Architecture:}
  \vspace{0.15in}
  \begin{itemize}
    \item \textbf{Corpus:} 6 ShopLite markdown policy documents in \texttt{data/kb/*.md}.
    \item \textbf{Chunking \& Embedding:} 600-character chunks with \texttt{paraphrase-multilingual-MiniLM-L12-v2}.
    \item \textbf{LLM Generation:} Qwen2.5-0.5B-Instruct (local CPU/GPU) or OpenAI-compatible endpoint.
    \item \textbf{Streaming Protocol:} Server-Sent Events (SSE) streaming token-by-token.
    \item \textbf{Retrieval Quality:} Hit@1 = 92.8\%, Hit@3 = 100.0\%.
  \end{itemize}
\end{minipage}
\hfill
\begin{minipage}[c]{0.48\textwidth}
  {\large Safety \& Citation Invariants:}
  \vspace{0.15in}
  \begin{itemize}
    \item \textbf{Source Attribution:} Mandatory citations formatted as \texttt{[file.md]}.
    \item \textbf{Fallback Behavior:} Fallback message when query is outside domain: ``Minh chua co thong tin nay, ban vui long lien he hotline...''.
    \item \textbf{Prompt-Injection Guard:} Strict system prompt instructing model to treat retrieved documents as data, not executable instructions.
  \end{itemize}
\end{minipage}
